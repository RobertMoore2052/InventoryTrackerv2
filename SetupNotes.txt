				public class InventoryItemViewModel

					public int InventoryItemId {get; set; }
					public string Name { get; set; } = "";
					public string? Description { get; set; }

					Web project needs a C# type into which that JSON can

					API
					InventoryItemDto
						|
						V JSON
					Web
					InventoryItemViewModel

		Replace contents of the .cshtml.cs file with:
			using InventoryTracker.Web.Models;
			using Microsoft.AspNetCore.Mvc.RazorPages;
			using System.Net.Http.Json;

			namespace InventoryTracker.Web.Pages.Inventory;

			public class IndexModel : PageModel
			{
				private readonly IHttpClientFactory _httpClientFactory;

				public List<InventoryItemViewModel> InventoryItems { get; set; } = [];

				public IndexModel(IHttpClientFactory httpClientFactory)
				{
					_httpClientFactory = httpClientFactory;
				}

				public async Task OnGetAsync()
				{
					var client = _httpClientFactory.CreateClient("InventoryTrackerApi");

					InventoryItems =
						await client.GetFromJsonAsync<List<InventoryItemViewModel>>(
							"api/InventoryItems")
						?? [];
				}
			}

		- NOTES ON THIS SECTION
			- IHttpClientFactory
				* Gives the page access to the client registered in Program.cs
			- _httpClientFactory.CreateClient("InventoryTrackerApi")
				* Retrieves that configured client
			- GetFromJsonAsync<List<InventoryItemViewModel>>(...)
				* does two things:
					- Send HTTP GET request
					- Deserialize returned JSON into C# objects
				* So this:
					"api/InventoryItems"
					- gets combined with your configured base address

			using Microsoft.AspNetCore.Mvc;
			using Microsoft.AspNetCore.Mvc.RazorPages;
			using System.Net.Http.Json;

			namespace InventoryTracker.Web.Pages.Inventory;

			public class DetailsModel : PageModel
			{
    			private readonly IHttpClientFactory _httpClientFactory;

    			public InventoryItemViewModel? InventoryItem { get; set; }

    			public DetailsModel(IHttpClientFactory httpClientFactory)
    			{
       			 _httpClientFactory = httpClientFactory;
    			}

    			public async Task<IActionResult> OnGetAsync(int id)
    			{
     			   var client = _httpClientFactory.CreateClient("InventoryTrackerApi");
        
     			   InventoryItem = 
     			       await client.GetFromJsonAsync<InventoryItemViewModel>(
     			           $"api/InventoryItem/{id}");

     			   if (InventoryItem == null)
     			   {
     			       return NotFound();
     			   }

     			   return Page();
    			}
			}

			*WHAT THIS DOES
				- User visits /Inventory/Details/12
					- razor supplies id = 12
					- HttpClient calls API /api/InventoryItems/12
					- API returns JSON
					- JSON becomes InventoryItemViewModel
					- Razor Page displays it
